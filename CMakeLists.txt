cmake_minimum_required(VERSION 3.20)
project(ascii-video-cpp VERSION 1.2.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
option(ASCII_BUILD_BENCHMARKS "Build benchmark and deterministic demo generator" ON)
find_package(Qt6 6.2 REQUIRED COMPONENTS Widgets)
find_package(OpenCV REQUIRED COMPONENTS core imgproc imgcodecs videoio)
if(OpenCV_VERSION VERSION_LESS "4.0" OR OpenCV_VERSION VERSION_GREATER_EQUAL "6.0")
  message(FATAL_ERROR "OpenCV 4.x or 5.x is required.")
endif()
find_package(Threads REQUIRED)
add_library(ascii_core src/AsciiConverter.cpp src/MediaInput.cpp src/MediaOutput.cpp
  src/FramePipeline.cpp src/GitHubExporter.cpp assets/fonts.qrc)
target_include_directories(ascii_core PUBLIC include)
target_include_directories(ascii_core SYSTEM PUBLIC ${OpenCV_INCLUDE_DIRS})
target_link_libraries(ascii_core PUBLIC Qt6::Gui ${OpenCV_LIBS} Threads::Threads)
function(ascii_warnings target)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /permissive- /utf-8)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
  endif()
endfunction()
ascii_warnings(ascii_core)
add_executable(ascii-video-cpp src/main.cpp src/MainWindow.cpp include/MainWindow.hpp assets/resources.qrc)
target_compile_definitions(ascii-video-cpp PRIVATE ASCII_VERSION="${PROJECT_VERSION}")
if(WIN32)
  target_sources(ascii-video-cpp PRIVATE assets/app.rc)
  set_target_properties(ascii-video-cpp PROPERTIES WIN32_EXECUTABLE TRUE)
endif()
target_link_libraries(ascii-video-cpp PRIVATE ascii_core Qt6::Widgets)
ascii_warnings(ascii-video-cpp)
include(CTest)
if(BUILD_TESTING)
  find_package(Qt6 REQUIRED COMPONENTS Test)
  add_executable(ascii-tests tests/tests.cpp src/MainWindow.cpp include/MainWindow.hpp assets/resources.qrc)
  target_link_libraries(ascii-tests PRIVATE ascii_core Qt6::Widgets Qt6::Test)
  ascii_warnings(ascii-tests)
  add_test(NAME ascii-tests COMMAND ascii-tests -o test-results.txt,txt -o test-results.xml,junitxml)
  set_tests_properties(ascii-tests PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 180)
endif()
if(ASCII_BUILD_BENCHMARKS)
  add_executable(ascii-benchmark benchmarks/main.cpp)
  target_link_libraries(ascii-benchmark PRIVATE ascii_core)
  ascii_warnings(ascii-benchmark)
endif()
if(APPLE)
  set_target_properties(ascii-video-cpp PROPERTIES MACOSX_BUNDLE TRUE
    MACOSX_BUNDLE_GUI_IDENTIFIER "io.github.dakiemdarktharr.ascii-video"
    MACOSX_BUNDLE_BUNDLE_NAME "ASCII Video" MACOSX_BUNDLE_BUNDLE_VERSION "${PROJECT_VERSION}"
    MACOSX_BUNDLE_SHORT_VERSION_STRING "${PROJECT_VERSION}" MACOSX_BUNDLE_ICON_FILE "app.icns")
  target_sources(ascii-video-cpp PRIVATE assets/icons/app.icns)
  set_source_files_properties(assets/icons/app.icns PROPERTIES MACOSX_PACKAGE_LOCATION Resources)
endif()
install(TARGETS ascii-video-cpp RUNTIME DESTINATION bin BUNDLE DESTINATION .)
if(UNIX AND NOT APPLE)
  install(FILES LICENSE assets/fonts/LICENSE-DejaVu.txt DESTINATION share/doc/ascii-video)
  install(FILES packaging/ascii-video.desktop DESTINATION share/applications)
  install(FILES assets/icons/app.png DESTINATION share/icons/hicolor/512x512/apps RENAME ascii-video.png)
  set(CPACK_GENERATOR "DEB")
  set(CPACK_PACKAGE_NAME "ascii-video")
  set(CPACK_PACKAGE_VERSION "${PROJECT_VERSION}")
  set(CPACK_PACKAGE_CONTACT "dakiemdarktharr <dakiemdarktharr@users.noreply.github.com>")
  set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "Convert videos and images into detailed ASCII art")
  set(CPACK_DEBIAN_PACKAGE_SHLIBDEPS ON)
  set(CPACK_DEBIAN_PACKAGE_DEPENDS "ffmpeg, fonts-dejavu-core, qt6-qpa-plugins")
  set(CPACK_DEBIAN_PACKAGE_MAINTAINER "dakiemdarktharr")
  set(CPACK_DEBIAN_FILE_NAME DEB-DEFAULT)
  include(CPack)
endif()
