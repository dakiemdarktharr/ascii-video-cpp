cmake_minimum_required(VERSION 3.20)
project(ascii-video-cpp VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_AUTOMOC ON)
option(ASCII_BUILD_BENCHMARKS "Build benchmark and deterministic demo generator" ON)
find_package(Qt6 6.2 REQUIRED COMPONENTS Widgets)
find_package(OpenCV REQUIRED COMPONENTS core imgproc imgcodecs videoio)
if(OpenCV_VERSION VERSION_LESS "4.0" OR OpenCV_VERSION VERSION_GREATER_EQUAL "6.0")
  message(FATAL_ERROR "OpenCV 4.x or 5.x is required.")
endif()
find_package(Threads REQUIRED)
add_library(ascii_core src/AsciiConverter.cpp src/MediaInput.cpp src/MediaOutput.cpp
  src/FramePipeline.cpp src/GitHubExporter.cpp)
target_include_directories(ascii_core PUBLIC include)
target_include_directories(ascii_core SYSTEM PUBLIC ${OpenCV_INCLUDE_DIRS})
target_link_libraries(ascii_core PUBLIC Qt6::Gui ${OpenCV_LIBS} Threads::Threads)
function(ascii_warnings target)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /permissive- /utf-8)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
  endif()
endfunction()
ascii_warnings(ascii_core)
add_executable(ascii-video-cpp src/main.cpp src/MainWindow.cpp include/MainWindow.hpp)
target_link_libraries(ascii-video-cpp PRIVATE ascii_core Qt6::Widgets)
ascii_warnings(ascii-video-cpp)
include(CTest)
if(BUILD_TESTING)
  find_package(Qt6 REQUIRED COMPONENTS Test)
  add_executable(ascii-tests tests/tests.cpp src/MainWindow.cpp include/MainWindow.hpp)
  target_link_libraries(ascii-tests PRIVATE ascii_core Qt6::Widgets Qt6::Test)
  ascii_warnings(ascii-tests)
  add_test(NAME ascii-tests COMMAND ascii-tests -o test-results.txt,txt -o test-results.xml,junitxml)
  set_tests_properties(ascii-tests PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 180)
endif()
if(ASCII_BUILD_BENCHMARKS)
  add_executable(ascii-benchmark benchmarks/main.cpp)
  target_link_libraries(ascii-benchmark PRIVATE ascii_core)
  ascii_warnings(ascii-benchmark)
endif()
install(TARGETS ascii-video-cpp RUNTIME DESTINATION bin)
